# Options reconnues par la version moOde de PeppyMeter.

# Les valeurs ci-dessous sont les valeurs exemple.

# ---------------------------------------------------------------------------
# config.txt, section [current] options standard de PeppyMeter
# ---------------------------------------------------------------------------
 meter = random                 # compteur unique, "random", ou liste "a,b,c" (option standard de PeppyMeter)
 random.meter.interval = 60     # secondes entre deux changements (option standard de PeppyMeter)
 random.change.title = False    # avec "random" ou une liste : changer de compteur a chaque nouveau morceau
 start.animation = False        # animation a l'ouverture et a la fermeture de la fenetre (SDL2)
 color.depth = 32
 position.type = center         "center", ou "custom" avec position.x et position.y
 position.x = 0
 position.y = 0
# ---------------------------------------------------------------------------
# extention de config.txt, section [current]
# ---------------------------------------------------------------------------
font.path = /usr/local/share/fonts	# répertoire des fonts
font.light = /Lato-Light.ttf		# nom du fichier font light
font.regular = /Lato-Regular.ttf	# nom du fichier font regular
font.bold = /Lato-Bold.ttf		# nom du fichier font bold

# ---------------------------------------------------------------------------
# meters.txt  dans la section de chaque vu-mètre options standard de PeppyMeter
# ---------------------------------------------------------------------------

font.size.digi = 40			# taille de la font digi en pixel (pour le temps restant)

# textes Album / titre / artiste 
playinfo.center	= True			# paramètre par défaut des informations (True :  centré )

playinfo.maxwidth = 616			# paramètre par défaut des informations : longueur max du texte (en pixel)
